dojoRng = function (seed) {
  let a = (seed * 7919 + 104729) >>> 0;
  return function () {
    a |= 0; a = (a + 0x6D2B79F5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

// FNV-1a — 학번 문자열을 정수로. 사람마다 다른 문제가 나오게 하는 출발점.
dojoHash = function (str) {
  let h = 2166136261 >>> 0;
  const s = String(str);
  for (let i = 0; i < s.length; i++) {
    h ^= s.charCodeAt(i);
    h = Math.imul(h, 16777619) >>> 0;
  }
  return h >>> 0;
}

dojoSeed = function (sid, week, type, round) {
  const key = (sid && String(sid).trim()) ? String(sid).trim() : "guest";

}

wR2 = (v) => Math.round(v * 100) / 100
wIoU = (A, B) => {
  const ix1 = Math.max(A.x, B.x), ix2 = Math.min(A.x + A.w, B.x + B.w);
  const iy1 = Math.max(A.y, B.y), iy2 = Math.min(A.y + A.h, B.y + B.h);
  const iw = Math.max(0, ix2 - ix1), ih = Math.max(0, iy2 - iy1);
  const inter = iw * ih, aA = A.w * A.h, aB = B.w * B.h, union = aA + aB - inter;
  return { ix1, ix2, iy1, iy2, iw, ih, inter, aA, aB, union, iou: inter / union };
}

  boxes.forEach(b => {

  const placed = [], hit = (a, c) => a.x < c.x + c.w && c.x < a.x + a.w && a.y < c.y + c.h && c.y < a.y + a.h;
  boxes.forEach(b => {

// NMS: boxes[i].conf, 짝마다 IoU 값 iou[i][j]. 기준 0.5 이상이면 지운다.
wNMS = (boxes, iou) => {
  let rest = boxes.map((_, i) => i).sort((a, b) => boxes[b].conf - boxes[a].conf);
  const steps = [];
  while (rest.length) {
    const top = rest.shift();
    const cmp = rest.map(j => ({ j, v: iou[top][j] }));
    const del = cmp.filter(c => c.v >= 0.5).map(c => c.j);
    rest = rest.filter(j => !del.includes(j));
    steps.push({ top, cmp, del });
  }
  return { steps, keep: steps.map(s => s.top) };
}

유형 G — 좌표로 판정하기 [도전]

viewof wG_round = Inputs.button("새 문제", { value: 0, reduce: (v) => v + 1 })

wG = {
  const rng = dojoRng(wG_seed);
  for (;;) {
    const g = { x: dojoInt(rng, 0, 30) * 10, y: dojoInt(rng, 0, 20) * 10, w: dojoInt(rng, 6, 20) * 10, h: dojoInt(rng, 6, 20) * 10 };
    const m = rng();
    const p = wNear(rng, g, m < 0.15 ? 1 : m < 0.6 ? 3 : 2);

    return { g, p, gc, pc, ...q, r: wR2(q.iou), ok: pc === gc && q.iou >= 0.5 };
  }
}

유형 H — 신뢰도로 거르고 NMS까지 [도전]

viewof wH_round = Inputs.button("새 문제", { value: 0, reduce: (v) => v + 1 })
wH_seed = dojoSeed(dojoSid, 9, "H", wH_round)
wH = {
  const rng = dojoRng(wH_seed);
  const hi = [0.95, 0.9, 0.85, 0.8, 0.75, 0.7, 0.65, 0.6, 0.55];
  const same = (a, b) => a.x === b.x && a.y === b.y && a.w === b.w && a.h === b.h;
  for (let t = 0; t < 5000; t++) {
    // 물체 두 개(왼쪽 절반, 오른쪽 절반)와 그 주변의 후보 박스
    const objs = [0, 1].map(i => {
      const w = dojoInt(rng, 3, 5), h = dojoInt(rng, 3, 5);
      return { x: i * 6 + dojoInt(rng, 0, 6 - w), y: dojoInt(rng, 0, 9 - h), w, h };
    });
    const counts = dojoPick(rng, [[2, 2], [3, 1], [1, 3], [2, 1], [1, 2]]);
    const raw = [];
    counts.forEach((n, oi) => {
      const o = objs[oi];
      for (let c = 0; c < n; c++) {
        const w = Math.max(2, o.w + dojoInt(rng, -1, 1)), h = Math.max(2, o.h + dojoInt(rng, -1, 1));
        const x = Math.min(12 - w, Math.max(0, o.x + dojoInt(rng, -1, 1))), y = Math.min(9 - h, Math.max(0, o.y + dojoInt(rng, -1, 1)));
        raw.push({ obj: oi, x, y, w, h });
      }
    });
    // 물체가 없는 자리의 후보 하나 — 신뢰도가 낮다
    const lw = dojoInt(rng, 2, 3), lh = dojoInt(rng, 2, 3);
    const lowBox = { obj: -1, x: dojoInt(rng, 0, 12 - lw), y: dojoInt(rng, 0, 9 - lh), w: lw, h: lh };
    if (raw.some(b => wIoU(b, lowBox).iou >= 0.2)) continue;
    raw.push(lowBox);
    for (let i = raw.length - 1; i > 0; i--) { const j = dojoInt(rng, 0, i); [raw[i], raw[j]] = [raw[j], raw[i]]; }
    const cs = [...hi];
    for (let i = cs.length - 1; i > 0; i--) { const j = dojoInt(rng, 0, i); [cs[i], cs[j]] = [cs[j], cs[i]]; }
    const boxes = raw.map((b, i) => ({ ...b, name: `B${i + 1}`, conf: b.obj < 0 ? dojoPick(rng, [0.2, 0.3, 0.35, 0.4, 0.45]) : cs[i] }));
    const n = boxes.length;
    const q = boxes.map(a => boxes.map(b => wIoU(a, b)));
    let bad = false;
    for (let i = 0; i < n; i++) for (let j = 0; j < n; j++) {
      if (i === j) continue;
      if (Math.abs(q[i][j].iou - 0.5) < 0.05 || same(boxes[i], boxes[j])) bad = true;
    }
    if (bad) continue;
    const idx = boxes.map((b, i) => i).filter(i => boxes[i].conf >= 0.5);
    const kept = idx.map(i => boxes[i]);
    const nms = wNMS(kept, idx.map(i => idx.map(j => q[i][j].iou)));
    // 물체마다 박스 하나씩 남아야 한다
    const ko = nms.keep.map(i => kept[i].obj);
    if (ko.length !== 2 || new Set(ko).size !== 2) continue;
    return { n, boxes, low: boxes.find(b => b.obj < 0), kept, q, idx, nms, keepNames: nms.keep.map(i => kept[i].name) };
  }
}
dojoBar(`H-${wH_seed}`)
dojoProblem(`문제 H — 후보 박스에서 최종 박스까지`, `
  탐지 모델이 후보 박스 ${wH.n}개를 냈다. 모눈 한 칸이 한 픽셀이다.
  ${wSvg(wH.boxes.map((b, i) => ({ ...b, color: wColors[i], label: b.name })), { grid: true, cols: 12, rows: 9, cs: 30, max: 440 })}
  <table style="border-collapse:collapse;font-size:0.9em;margin:0.3em 0;width:auto;">
    <tr>${["박스", "(x, y, w, h)", "신뢰도"].map(t => `<th style="padding:0.2em 0.7em;background:#8C1515;color:#fff;">${t}</th>`).join("")}</tr>
    ${wH.boxes.map(b => `<tr>${[b.name, `(${b.x}, ${b.y}, ${b.w}, ${b.h})`, b.conf.toFixed(2)].map(t => `<td style="padding:0.2em 0.7em;border:1px solid #ddd;text-align:center;">${t}</td>`).join("")}</tr>`).join("")}
  </table>
  신뢰도가 <strong>0.5 미만</strong>인 후보를 먼저 버리고, 남은 박스에 NMS(기준 IoU 0.5 이상)를 적용한다.
  최종으로 남는 박스를 모두 고르라.`)
viewof wH_ans = Inputs.checkbox(wH.boxes.map(b => b.name), { label: "남는 박스" })
viewof wH_grade = Inputs.toggle({ label: "채점", value: false })
viewof wH_show = Inputs.toggle({ label: "풀이 보기", value: false })
wH_grade ? dojoResult([dojoMark("남는 박스", wSameSet(wH_ans, wH.keepNames))], wSameSet(wH_ans, wH.keepNames)) : html``
wH_show ? dojoSolution(`
  ① 신뢰도 0.5 미만인 <strong>${wH.low.name}</strong>(${wH.low.conf.toFixed(2)})를 버린다.
  남은 박스: ${wH.kept.map(b => b.name).join(", ")}<br>
  ② 남은 박스로 NMS를 한다. 필요한 IoU는 교집합 ÷ 합집합으로 계산한다.
  ${wNMSTable(wH.kept, wH.nms, (i, j) => {
    const r = wH.q[wH.idx[i]][wH.idx[j]];
    return `${r.inter}/${r.union} = ${wR2(r.iou).toFixed(2)}`;
  })}
  <span style="color:#555;">신뢰도가 낮은 후보를 버리면 물체가 있는 자리의 박스만 남고,
  NMS는 같은 물체를 가리키는 박스 가운데 신뢰도가 가장 높은 것만 남긴다.</span>`) : html``
