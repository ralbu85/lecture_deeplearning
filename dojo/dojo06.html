// mulberry32 — 문제 번호(시드)가 같으면 항상 같은 문제가 생성된다
dojoRng = function (seed) {
  let a = (seed * 7919 + 104729) >>> 0;
  return function () {
    a |= 0; a = (a + 0x6D2B79F5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
dojoInt = (rng, lo, hi) => lo + Math.floor(rng() * (hi - lo + 1))
dojoPick = (rng, arr) => arr[Math.floor(rng() * arr.length)]

  user !== null && user !== undefined && Number.isFinite(user) && Math.abs(user - answer) <= tol

// FNV-1a — 학번 문자열을 정수로. 사람마다 다른 문제가 나오게 하는 출발점.
dojoHash = function (str) {
  let h = 2166136261 >>> 0;
  const s = String(str);
  for (let i = 0; i < s.length; i++) {
    h ^= s.charCodeAt(i);
    h = Math.imul(h, 16777619) >>> 0;
  }
  return h >>> 0;
}
// (학번, 주차, 유형, 라운드) → 문제 번호.
// 학번이 다르면 다른 문제, 같은 학번·같은 라운드면 항상 같은 문제(재현 가능).
dojoSeed = function (sid, week, type, round) {
  const key = (sid && String(sid).trim()) ? String(sid).trim() : "guest";
  return (dojoHash(key + "|" + week + "|" + type + "|" + round) % 99999) + 1;
}

// 음수를 −(U+2212)로 표시
dojoFmt = (v) => typeof v === "number" ? (v < 0 ? "−" + String(-v) : String(v)) : v
// 행렬을 칸 표로 — 문제·풀이 본문에 넣는 HTML 문자열
// opt.hl(r, c): 배경색 또는 null, opt.dim(r, c): true면 점선·회색 칸(패딩), opt.label: 위쪽 이름
dojoMat = (M, opt = {}) => {
  const hl = opt.hl || (() => null), dim = opt.dim || (() => false), f = opt.fmt || dojoFmt;
  const body = M.map((row, r) => "<tr>" + row.map((v, c) => {
    const d = dim(r, c), bg = hl(r, c);
    return `<td style="min-width:2.1em;height:2.1em;padding:0 0.35em;text-align:center;` +
      `border:1px ${d ? "dashed #bbb" : "solid #999"};color:${d ? "#999" : "#222"};` +
      `background:${bg || "#fff"};font-variant-numeric:tabular-nums;">${f(v)}</td>`;
  }).join("") + "</tr>").join("");
  const cap = opt.label ? `<div style="font-size:0.85em;color:#555;margin-bottom:0.15em;">${opt.label}</div>` : "";
  return `<div style="display:inline-block;vertical-align:top;margin:0.3em 1.2em 0.3em 0;">${cap}` +
    `<table style="border-collapse:collapse;width:auto;margin:0;">${body}</table></div>`;
}
// 행렬 모양의 답 입력 칸 — 값은 행 순서로 펼친 배열(빈 칸은 null)
dojoGrid = (rows, cols, label = "") => {
  const wrap = document.createElement("div");
  wrap.style.cssText = "font-family:'Pretendard',sans-serif;margin:0.3em 0 0.6em;display:flex;align-items:center;gap:0.8em;";
  if (label) {
    const l = document.createElement("div");
    l.textContent = label;
    l.style.cssText = "font-size:0.9em;min-width:6.5em;";
    wrap.appendChild(l);
  }
  const t = document.createElement("table");
  t.style.cssText = "border-collapse:collapse;width:auto;margin:0;";
  const cells = [];
  for (let r = 0; r < rows; r++) {
    const tr = document.createElement("tr");
    for (let c = 0; c < cols; c++) {
      const td = document.createElement("td");
      td.style.cssText = "padding:2px;border:none;";
      const inp = document.createElement("input");
      inp.type = "number"; inp.step = "any";
      inp.style.cssText = "width:4.6em;text-align:center;padding:0.2em;border:1px solid #bbb;border-radius:4px;font-size:0.95em;";
      td.appendChild(inp); tr.appendChild(td); cells.push(inp);
    }
    t.appendChild(tr);
  }
  wrap.appendChild(t);
  Object.defineProperty(wrap, "value", { get: () => cells.map(i => i.value === "" ? null : Number(i.value)) });
  return wrap;
}

dojoCount = (users, answers, tol = 0.005) =>

// 콘볼루션 한 칸: 입력 행렬 X의 (i, j)에서 시작하는 K×K 부분과 커널의 스칼라 곱
tConvAt = (X, ker, i, j) => {
  const terms = [];
  for (let r = 0; r < ker.length; r++)
    for (let c = 0; c < ker[0].length; c++) terms.push([X[i + r][j + c], ker[r][c]]);
  return { terms, v: terms.reduce((a, [x, k]) => a + x * k, 0) };
}
// 곱의 합을 풀어 쓴 문자열: 1·0 + 2·(−1) + …
tTerms = (terms) => terms.map(([x, k]) => `${dojoFmt(x)}·${k < 0 ? "(" + dojoFmt(k) + ")" : k}`).join(" + ")
tRandMat = (rng, n, m, vals) => Array.from({ length: n }, () => Array.from({ length: m }, () => dojoPick(rng, vals)))

유형 C — 특성 맵의 크기 공식 [기본]

viewof tC_round = Inputs.button("새 문제", { value: 0, reduce: (v) => v + 1 })
tC_seed = dojoSeed(dojoSid, 6, "C", tC_round)
tC = {
  const rng = dojoRng(tC_seed);
  if (rng() < 0.75) {
    let N, K, P, S;
    do {
      N = dojoPick(rng, [5, 7, 9, 12, 14, 16, 28, 32, 39, 48, 55, 64, 100, 128, 224, 227]);
      K = dojoPick(rng, [2, 3, 3, 4, 5, 5, 7, 11]);
      P = dojoPick(rng, [0, 0, 1, 1, 2, 3]);
      S = dojoPick(rng, [1, 1, 2, 2, 3, 4]);
    } while (N + 2 * P < K);
    const num = N - K + 2 * P;
    return { kind: 0, N, K, P, S, num, out: Math.floor(num / S) + 1, ans: Math.floor(num / S) + 1 };
  }
  const K = dojoPick(rng, [3, 5, 7, 9, 11]);
  const N = dojoPick(rng, [16, 28, 32, 64, 224]);
  return { kind: 1, N, K, S: 1, ans: (K - 1) / 2 };
}

    + 채널 2 [${tTerms(c.ch[1].terms)} = ${dojoFmt(c.ch[1].v)}] + (${dojoFmt(tH.b)}) = <strong>${dojoFmt(c.v)}</strong>`).join("<br>")}<br>
