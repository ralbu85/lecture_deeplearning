// mulberry32 — 문제 번호(시드)가 같으면 항상 같은 문제가 생성된다
dojoRng = function (seed) {
  let a = (seed * 7919 + 104729) >>> 0;
  return function () {
    a |= 0; a = (a + 0x6D2B79F5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

// FNV-1a — 학번 문자열을 정수로. 사람마다 다른 문제가 나오게 하는 출발점.
dojoHash = function (str) {
  let h = 2166136261 >>> 0;
  const s = String(str);
  for (let i = 0; i < s.length; i++) {
    h ^= s.charCodeAt(i);
    h = Math.imul(h, 16777619) >>> 0;
  }
  return h >>> 0;
}

dojoSeed = function (sid, week, type, round) {

// 층 배치 생성 — tpl 0: 콘볼루션·풀링 두 번, tpl 1: 콘볼루션 세 번(뒤 두 층 스트라이드 2)
uGenArch = (rng, tpl) => {
  for (let tries = 0; tries < 500; tries++) {
    if (tpl === 0) {
      const N = dojoPick(rng, [28, 32, 36, 40, 48, 64]);
      const C0 = dojoPick(rng, [1, 3, 3]);
      const K1 = dojoPick(rng, [3, 5]), P1 = dojoPick(rng, [0, (K1 - 1) / 2]), M1 = dojoPick(rng, [6, 8, 16, 32]);
      const o1 = N - K1 + 2 * P1 + 1;
      if (o1 % 2) continue;
      const p1 = o1 / 2;
      const K2 = dojoPick(rng, [3, 5]), P2 = dojoPick(rng, [0, (K2 - 1) / 2]), M2 = dojoPick(rng, [10, 16, 32, 64]);
      const o2 = p1 - K2 + 2 * P2 + 1;
      if (o2 % 2 || o2 < 2) continue;
      const p2 = o2 / 2;
      const layers = [
        { name: "Conv1", type: "conv", K: K1, S: 1, P: P1, M: M1, nin: N, cin: C0, out: o1, ch: M1 },
        { name: "Pool1", type: "pool", K: 2, S: 2, nin: o1, cin: M1, out: p1, ch: M1 },
        { name: "Conv2", type: "conv", K: K2, S: 1, P: P2, M: M2, nin: p1, cin: M1, out: o2, ch: M2 },
        { name: "Pool2", type: "pool", K: 2, S: 2, nin: o2, cin: M2, out: p2, ch: M2 }
      ];
      return { tpl, N, C0, layers, last: p2, lastC: M2, flat: p2 * p2 * M2 };
    } else {
      const N = dojoPick(rng, [32, 39, 48, 63, 64]);
      const C0 = 3;
      const spec = [
        { K: dojoPick(rng, [3, 5]), S: 1, M: dojoPick(rng, [8, 10, 16]) },
        { K: dojoPick(rng, [3, 5]), S: 2, M: dojoPick(rng, [16, 20, 32]) },
        { K: dojoPick(rng, [3, 5]), S: 2, M: dojoPick(rng, [32, 40, 64]) }
      ];
      let n = N, cin = C0;
      const layers = [];
      spec.forEach((s, i) => {
        const out = Math.floor((n - s.K) / s.S) + 1;
        layers.push({ name: `Conv${i + 1}`, type: "conv", K: s.K, S: s.S, P: 0, M: s.M, nin: n, cin, out, ch: s.M });
        n = out; cin = s.M;
      });
      if (n < 2) continue;
      return { tpl, N, C0, layers, last: n, lastC: cin, flat: n * n * cin };
    }
  }
}
// 파라미터 수를 붙인다 (콘볼루션: (K·K·C + 1)·M, 풀링: 0, FC: n_in·n_out + n_out)
uAddParams = (a, h1, cls) => {
  a.layers.forEach(l => { l.par = l.type === "conv" ? (l.K * l.K * l.cin + 1) * l.M : 0; });
  a.h1 = h1; a.cls = cls;
  a.fc1 = a.flat * h1 + h1;
  a.fc2 = h1 * cls + cls;
  a.featPar = a.layers.reduce((s, l) => s + l.par, 0);
  a.total = a.featPar + a.fc1 + a.fc2;
  return a;
}

uA = {
  const rng = dojoRng(uA_seed);
  const kind = rng() < 0.65 ? "Max" : "Average";
  const M = Array.from({ length: 4 }, () => Array.from({ length: 4 }, () => dojoInt(rng, -9, 9)));
  if (rng() < 0.6) {
    const w = dojoInt(rng, 0, 3), wr = Math.floor(w / 2) * 2, wc = (w % 2) * 2;
    for (let r = 0; r < 2; r++) for (let c = 0; c < 2; c++) M[wr + r][wc + c] = -dojoInt(rng, 1, 9);
  }
  const R = M.map(row => row.map(v => Math.max(0, v)));
  const wins = [];
  for (let r = 0; r < 2; r++) for (let c = 0; c < 2; c++) {
    const vals = [R[2 * r][2 * c], R[2 * r][2 * c + 1], R[2 * r + 1][2 * c], R[2 * r + 1][2 * c + 1]];
    const v = kind === "Max" ? Math.max(...vals) : vals.reduce((a, b) => a + b, 0) / 4;
    wins.push({ r, c, vals, v });
  }
  return { kind, M, R, wins, out: wins.map(w => w.v) };
}

  <span style="color:#666;">(②③ 소수 넷째 자리까지)</span>`)
